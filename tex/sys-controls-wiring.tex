% ── Critical Invariants Box ────────────────────────────────────────────────
\begin{tcolorbox}[bwalert]
\textbf{CORE ELECTRICAL INVARIANTS \& CONNECTION RULES:}
\begin{itemize}[leftmargin=1.3em, itemsep=1pt, topsep=1pt, parsep=0pt]
  \item \textbf{Active-LOW Inputs:} Every digital input connects directly between the assigned \textbf{GPIO} and \textbf{Common GND (0\,V)}. Internal MCU pull-up resistors ($\approx$3.3\,V) are active on all input pins.
  \item \textbf{Strict Voltage Limit:} \textbf{NEVER connect +12\,V or external 3.3\,V to any GPIO pin.} Doing so destroys the ESP32-S3 SoC immediately.
  \item \textbf{Fail-Safe ESTOP Loop:} Must use a \textbf{Normally Closed (NC)} contact to GND. Disconnection, broken wire, or button push opens circuit $\rightarrow$ pulls to 3.3\,V $\rightarrow$ latches ESTOP.
  \item \textbf{Wiring Standard:} Use \textbf{22 AWG stranded wire} for all switch lines and CAN bus pairs.
\end{itemize}
\end{tcolorbox}

\vspace{-5pt}

% ── USB Pin Conflict Alert Box ─────────────────────────────────────────────
\begin{tcolorbox}[bwalert]
\textbf{NATIVE USB PIN CONFLICT WARNING (GPIO 19 \& GPIO 20):}
GPIO 19 and GPIO 20 are physically shared with the ESP32-S3 Native USB data lines (D- / D+). \textbf{Relay Channels K4 and K7 must remain completely disconnected (NO, COM, NC all open).} Connecting loads to these channels causes severe relay chatter whenever USB is attached.
\end{tcolorbox}
